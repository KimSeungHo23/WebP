const SPREADSHEET_ID = "1a9iHMJIKy3lNpY41yQFhJemBDOSVCz8uWT1VLn4mZSw";
const SHEET_NAME = "시트1";

function doGet() {
  return HtmlService.createHtmlOutputFromFile("index")
    .setTitle("11월25일 저녁 신청 받기");
}

function submitForm(data) {
  const sheet = SpreadsheetApp
    .openById(SPREADSHEET_ID)
    .getSheetByName(SHEET_NAME);

  if (!sheet) {
    throw new Error("시트 이름을 확인해주세요.");
  }

  if (!data.name || !data.food) {
    throw new Error("이름과 음식을 입력해주세요.");
  }

  sheet.appendRow([
    data.name,
    data.food,
    data.message
  ]);

  return "답장 감사합니다.";
}
